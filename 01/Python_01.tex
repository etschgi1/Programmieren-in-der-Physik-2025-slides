  \begin{center}

  \end{center}

  \vspace{5mm}
  
  \textbf{Programming:}  
  \begin{lstlisting}[language=Python]

x = x + 1
print(x)  # Output: 6
\end{lstlisting}
\begin{center}
  \textcolor{red}{\large\( x = 5 \Rightarrow\) value of  $x$ can be changed}
